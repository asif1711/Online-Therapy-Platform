2. 'users' Table Schema :
CREATE TABLE users (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    uuid VARCHAR2(36) UNIQUE NOT NULL,
    full_name VARCHAR2(100) NOT NULL,
    email VARCHAR2(150) UNIQUE NOT NULL,
    password_hash VARCHAR2(255) NOT NULL,
    role VARCHAR2(10) CHECK (role IN ('ADMIN','USER')),
    status VARCHAR2(10) CHECK (status IN ('ACTIVE','INACTIVE','BANNED')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login_at TIMESTAMP
);

4. 'therapists' Table Schema :
CREATE TABLE therapists (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    uuid VARCHAR2(36) UNIQUE NOT NULL,
    full_name VARCHAR2(100) NOT NULL,
    title VARCHAR2(150),
    bio CLOB,
    experience_years NUMBER(2),
    image_url VARCHAR2(255),
    rating NUMBER(2,1) DEFAULT 0.0,
    review_count NUMBER DEFAULT 0,
    is_active CHAR(1) DEFAULT 'Y' CHECK (is_active IN ('Y','N')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP
);

5. 'specialties' Table Schema :
CREATE TABLE specialties (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR2(100) UNIQUE NOT NULL
);

6. 'therapist_specialties' Table Schema :
CREATE TABLE therapist_specialties (
    therapist_id NUMBER NOT NULL,
    specialty_id NUMBER NOT NULL,
    CONSTRAINT pk_therapist_specialties PRIMARY KEY (therapist_id, specialty_id),
    CONSTRAINT fk_ts_therapist
        FOREIGN KEY (therapist_id) REFERENCES therapists(id) ON DELETE CASCADE,
    CONSTRAINT fk_ts_specialty
        FOREIGN KEY (specialty_id) REFERENCES specialties(id) ON DELETE CASCADE
);

7. 'appointments' Table Schema :
CREATE TABLE appointments (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    uuid VARCHAR2(36) UNIQUE NOT NULL,
    patient_name VARCHAR2(100) NOT NULL,
    patient_email VARCHAR2(150) NOT NULL,
    patient_phone VARCHAR2(15) NOT NULL,
    therapist_id NUMBER,
    appointment_type VARCHAR2(20) NOT NULL CHECK (appointment_type IN ('video','in-person','phone')),
    preferred_date DATE NOT NULL,
    preferred_time VARCHAR2(5) NOT NULL,
    reason CLOB,
    status VARCHAR2(20) DEFAULT 'PENDING' CHECK (status IN ('PENDING','CONFIRMED','CANCELLED','COMPLETED')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_appointments_therapist
        FOREIGN KEY (therapist_id)
        REFERENCES therapists(id)
        ON DELETE SET NULL
);

8. 'contact_messages' Table Schema :
CREATE TABLE contact_messages (
id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
uuid VARCHAR2(36) UNIQUE NOT NULL,
name VARCHAR2(100) NOT NULL,
email VARCHAR2(150) NOT NULL,
phone VARCHAR2(15),
subject VARCHAR2(200) NOT NULL,
message CLOB NOT NULL,
status VARCHAR2(20) DEFAULT 'NEW' CHECK (status IN ('NEW','READ','REPLIED','ARCHIVED')),
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
